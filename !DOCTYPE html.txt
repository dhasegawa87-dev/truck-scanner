  <script>
    // ★ここに先ほどコピーしたGASのウェブアプリURLを貼り付けてください
    const GAS_URL = "YOUR_GAS_WEB_APP_URL";https://script.google.com/macros/s/AKfycbxUkUHjDa7dCj5c6LJdjH0UNPfCbPSHQAwfUUxopKle3GvEWitIt10qUE4RAHFZDu3gbQ/exec

    let currentTruckId = "";
    let html5QrCode;
    let isProcessing = false;

    window.onload = function() {
      html5QrCode = new Html5Qrcode("reader");
      startCamera();
    };

    function startCamera() {
      const config = { fps: 10, qrbox: { width: 250, height: 250 } };
      html5QrCode.start(
        { facingMode: "environment" },
        config,
        onScanSuccess
      ).catch(err => {
        alert("カメラの起動に失敗しました。ブラウザのカメラ権限を許可してください。");
      });
    }

    function onScanSuccess(decodedText) {
      if (isProcessing) return;
      isProcessing = true;

      if (!currentTruckId) {
        currentTruckId = decodedText;
        document.getElementById("disp-truck-id").innerText = currentTruckId;
        document.getElementById("step-label").innerText = "Step 2: かご車IDを連続スキャンしてください";
        document.getElementById("step-label").style.background = "#e6f4ea";
        document.getElementById("step-label").style.color = "#137333";
        document.getElementById("reset-truck-btn").style.display = "block";
        
        if (navigator.vibrate) navigator.vibrate(200);
        setTimeout(() => { isProcessing = false; }, 1500);
      } else {
        const cageId = decodedText;
        
        // GASへデータを送信
        fetch(GAS_URL, {
          method: "POST",
          mode: "no-cors", // GASへのクロスドメイン対策
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ truckId: currentTruckId, cageId: cageId })
        }).then(() => {
          const log = document.getElementById("scan-log");
          const li = document.createElement("li");
          const now = new Date().toLocaleTimeString();
          li.innerText = `[${now}] かご車: ${cageId}`;
          log.insertBefore(li, log.firstChild);
          
          if (navigator.vibrate) navigator.vibrate([100, 50, 100]);
          setTimeout(() => { isProcessing = false; }, 1200);
        }).catch(err => {
          alert("送信エラーが発生しました");
          isProcessing = false;
        });
      }
    }

    function resetTruck() {
      currentTruckId = "";
      document.getElementById("disp-truck-id").innerText = "未設定";
      document.getElementById("step-label").innerText = "Step 1: トラック便IDをスキャンしてください";
      document.getElementById("step-label").style.background = "#eef6ff";
      document.getElementById("step-label").style.color = "#1a73e8";
      document.getElementById("reset-truck-btn").style.display = "none";
      document.getElementById("scan-log").innerHTML = "";
    }
  </script>
